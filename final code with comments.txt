import os
import csv
import cv2
import numpy as np
import matplotlib.pyplot as plt
import matplotlib.dates as mdates
from datetime import datetime, timedelta
from matplotlib.gridspec import GridSpec

class LEDLifespanEstimator:
    def __init__(self, images_dir, output_dir="results"):
        """
        Initialize the LED Lifespan Estimator.
        
        Args:
            images_dir (str): Directory containing LED bulb images
            output_dir (str): Directory to save results
        """
        self.images_dir = images_dir
        self.output_dir = output_dir
        self.reference_brightness = None
        self.reference_color_temp = None
        self.history = []
        self.timestamps = []
        
        # Create output directory if it doesn't exist
        if not os.path.exists(output_dir):
            os.makedirs(output_dir)
            
        # Create subdirectory for step-by-step images
        self.steps_dir = os.path.join(output_dir, "processing_steps")
        if not os.path.exists(self.steps_dir):
            os.makedirs(self.steps_dir)
            
        # Parameters for lifespan estimation
        self.brightness_threshold = 0.7  # 70% of initial brightness
        self.color_temp_threshold = 0.15  # 15% shift in color temperature
        self.flicker_threshold = 0.1  # 10% increase in flicker
        
        # Feature weights for overall health score
        self.weights = {
            'brightness': 0.6,
            'color_temp': 0.3,
            'flicker': 0.1
        }

    def load_images(self):
        """
        Load images from the specified directory and sort them by name.
        
        Returns:
            list: List of image file paths sorted by name
        """
        image_files = [f for f in os.listdir(self.images_dir) 
                      if f.endswith(('.jpg', '.jpeg', '.png','DNG'))]
        image_files.sort()  # Sort by filename
        return [os.path.join(self.images_dir, f) for f in image_files]

    def save_step_image(self, image, step_name, image_index, original_filename):
        """
        Save an image from a processing step.
        
        Args:
            image (numpy.ndarray): Image to save
            step_name (str): Name of the processing step
            image_index (int): Index of the current image being processed
            original_filename (str): Original filename for reference
        """
        # Create filename for the step
        base_name = os.path.splitext(original_filename)[0]
        step_filename = f"{base_name}_{image_index:02d}_{step_name}.png"
        step_path = os.path.join(self.steps_dir, step_filename)
        
        # Save the image
        cv2.imwrite(step_path, image)
        print(f"    Saved {step_name}: {step_filename}")

    def create_visualization_image(self, original, masked, hsv, lab, gray, mask, image_index, original_filename):
        """
        Create a visualization showing all processing steps in one image.
        
        Args:
            original (numpy.ndarray): Original image
            masked (numpy.ndarray): Masked image
            hsv (numpy.ndarray): HSV image
            lab (numpy.ndarray): LAB image
            gray (numpy.ndarray): Grayscale image
            mask (numpy.ndarray): Binary mask
            image_index (int): Index of the current image
            original_filename (str): Original filename
        """
        # Create a composite image showing key steps
        h, w = original.shape[:2]
        
        # Create a 2x3 grid layout
        composite = np.zeros((h * 2, w * 3, 3), dtype=np.uint8)
        
        # Add labels and images
        steps = [
            (original, "1. Original"),
            (masked, "2. Masked"),
            (cv2.cvtColor(hsv, cv2.COLOR_HSV2BGR), "3. HSV"),
            (cv2.cvtColor(lab, cv2.COLOR_LAB2BGR), "4. LAB"),
            (cv2.cvtColor(gray, cv2.COLOR_GRAY2BGR), "5. Grayscale"),
            (cv2.cvtColor((mask * 255).astype(np.uint8), cv2.COLOR_GRAY2BGR), "6. Adaptive Mask")
        ]
        
        for i, (img, label) in enumerate(steps):
            row = i // 3
            col = i % 3
            
            # Add the image
            composite[row*h:(row+1)*h, col*w:(col+1)*w] = img
            
            # Add label
            cv2.putText(composite, label, (col*w + 10, row*h + 30), 
                       cv2.FONT_HERSHEY_SIMPLEX, 0.8, (0, 255, 0), 2)
        
        # Save the composite image
        base_name = os.path.splitext(original_filename)[0]
        composite_filename = f"{base_name}_{image_index:02d}_all_steps.png"
        composite_path = os.path.join(self.steps_dir, composite_filename)
        cv2.imwrite(composite_path, composite)
        print(f"    Saved composite: {composite_filename}")

    
    def create_adaptive_mask(self, img, image_index, original_filename):
        """
        Create an adaptive mask for the LED bulb based on brightness and color.
        
        Args:
            img (numpy.ndarray): Input image
            image_index (int): Index of the current image
            original_filename (str): Original filename
            
        Returns:
            numpy.ndarray: Binary mask for the LED bulb area
        """
        # Convert to grayscale for initial processing
        gray = cv2.cvtColor(img, cv2.COLOR_BGR2GRAY)
        self.save_step_image(gray, "02_grayscale_initial", image_index, original_filename)
        
        # Apply Gaussian blur to reduce noise
        blurred = cv2.GaussianBlur(gray, (11, 11), 0)
        self.save_step_image(blurred, "03_blurred", image_index, original_filename)
        
        # Use adaptive thresholding to find bright regions
        # First, find the brightest areas using a high threshold
        bright_threshold = np.max(gray) * 0.6  # 60% of maximum brightness
        _, bright_mask = cv2.threshold(blurred, bright_threshold, 255, cv2.THRESH_BINARY)
        self.save_step_image(bright_mask, "04_bright_threshold", image_index, original_filename)
        
        # Use Otsu's thresholding as a backup method
        _, otsu_mask = cv2.threshold(blurred, 0, 255, cv2.THRESH_BINARY + cv2.THRESH_OTSU)
        self.save_step_image(otsu_mask, "05_otsu_threshold", image_index, original_filename)
        
        # Combine both masks
        combined_mask = cv2.bitwise_or(bright_mask, otsu_mask)
        self.save_step_image(combined_mask, "06_combined_mask", image_index, original_filename)
        
        # Apply morphological operations to clean up the mask
        kernel = np.ones((5, 5), np.uint8)
        
        # Remove small noise
        cleaned_mask = cv2.morphologyEx(combined_mask, cv2.MORPH_OPEN, kernel)
        self.save_step_image(cleaned_mask, "07_opened_mask", image_index, original_filename)
        
        # Fill holes
        filled_mask = cv2.morphologyEx(cleaned_mask, cv2.MORPH_CLOSE, kernel)
        self.save_step_image(filled_mask, "08_filled_mask", image_index, original_filename)
        
        # Find contours to identify the main LED bulb
        contours, _ = cv2.findContours(filled_mask, cv2.RETR_EXTERNAL, cv2.CHAIN_APPROX_SIMPLE)
        
        # Create image to visualize contours
        contour_img = img.copy()
        cv2.drawContours(contour_img, contours, -1, (0, 255, 0), 2)
        self.save_step_image(contour_img, "09_contours", image_index, original_filename)
        
        # Find the largest contour (should be the main LED bulb)
        if contours:
            largest_contour = max(contours, key=cv2.contourArea)
            
            # Create mask from the largest contour
            final_mask = np.zeros(gray.shape, dtype=np.uint8)
            cv2.fillPoly(final_mask, [largest_contour], 255)
            
            # Apply additional filtering based on contour properties
            area = cv2.contourArea(largest_contour)
            if area > 100:  # Minimum area threshold
                # Get bounding rectangle
                x, y, w, h = cv2.boundingRect(largest_contour)
                aspect_ratio = float(w) / h
                
                # If the aspect ratio is reasonable for a bulb (roughly circular)
                if 0.5 < aspect_ratio < 2.0:
                    # Optionally, create a more refined mask using the convex hull
                    hull = cv2.convexHull(largest_contour)
                    cv2.fillPoly(final_mask, [hull], 255)
        else:
            # Fallback: use the filled mask if no good contours found
            final_mask = filled_mask
            
        # Apply additional smoothing to the final mask
        final_mask = cv2.GaussianBlur(final_mask, (5, 5), 0)
        _, final_mask = cv2.threshold(final_mask, 127, 255, cv2.THRESH_BINARY)
        
        self.save_step_image(final_mask, "10_final_mask", image_index, original_filename)
        
        # Create visualization showing mask overlay - FIXED VERSION
        mask_overlay = img.copy()
        
        # Create a green overlay for the masked regions
        green_overlay = np.zeros_like(img)
        green_overlay[final_mask > 0] = [0, 255, 0]  # Green color
        
        # Blend the original image with the green overlay
        mask_overlay = cv2.addWeighted(mask_overlay, 0.7, green_overlay, 0.3, 0)
        
        self.save_step_image(mask_overlay, "11_mask_overlay", image_index, original_filename)
        
        return final_mask > 0  # Return boolean mask

    def extract_features(self, image_path, image_index=0):
        """
        Extract brightness, color temperature, and flicker features from an image.
        
        Args:
            image_path (str): Path to the image file
            image_index (int): Index of the current image being processed
            
        Returns:
            dict: Dictionary containing extracted features
        """
        # Read the image
        img = cv2.imread(image_path)
        if img is None:
            raise ValueError(f"Could not read image at {image_path}")
        
        original_filename = os.path.basename(image_path)
        print(f"  Processing steps for {original_filename}:")
        
        # Save original image
        self.save_step_image(img, "01_original", image_index, original_filename)
        
        # --- ADAPTIVE MASK CREATION ---
        mask = self.create_adaptive_mask(img, image_index, original_filename)
        mask_3ch = np.stack([mask] * 3, axis=-1)  # For color images

        # Save final mask
        mask_img = (mask * 255).astype(np.uint8)
        self.save_step_image(mask_img, "12_adaptive_mask", image_index, original_filename)

        # Apply mask to the image
        img_masked = img.copy()
        img_masked[~mask_3ch] = 0  # Set background to black
        
        # Save masked image
        self.save_step_image(img_masked, "13_masked", image_index, original_filename)

        # Convert to HSV and LAB color spaces
        hsv_img = cv2.cvtColor(img_masked, cv2.COLOR_BGR2HSV)
        lab_img = cv2.cvtColor(img_masked, cv2.COLOR_BGR2LAB)
        gray_img = cv2.cvtColor(img_masked, cv2.COLOR_BGR2GRAY)
        
        # Save converted images
        self.save_step_image(hsv_img, "14_hsv", image_index, original_filename)
        self.save_step_image(lab_img, "15_lab", image_index, original_filename)
        self.save_step_image(gray_img, "16_grayscale", image_index, original_filename)
        
        # Extract brightness (V channel from HSV)
        brightness = np.mean(hsv_img[:, :, 2][mask]) / 255.0
        
        # Save brightness channel
        brightness_img = hsv_img[:, :, 2].copy()
        brightness_img[~mask] = 0
        self.save_step_image(brightness_img, "17_brightness_channel", image_index, original_filename)
        
        # Extract color temperature (using a/b channels from LAB)
        # This is a simplified estimation - lower values indicate cooler temperatures
        a_channel = np.mean(lab_img[:, :, 1][mask])
        b_channel = np.mean(lab_img[:, :, 2][mask])
        color_temp = np.sqrt(a_channel**2 + b_channel**2)
        
        # Save color temperature visualization
        a_img = lab_img[:, :, 1].copy()
        b_img = lab_img[:, :, 2].copy()
        a_img[~mask] = 128  # Set to neutral
        b_img[~mask] = 128  # Set to neutral
        self.save_step_image(a_img, "18_a_channel", image_index, original_filename)
        self.save_step_image(b_img, "19_b_channel", image_index, original_filename)
        
        # If this is the first image, set reference values
        if self.reference_brightness is None:
            self.reference_brightness = brightness
            self.reference_color_temp = color_temp
            flicker_index = 0.0
            diff_img = np.zeros_like(gray_img)
        else:
            # Calculate flicker index (difference between consecutive frames)
            if hasattr(self, 'prev_gray'):
                diff_img = cv2.absdiff(gray_img, self.prev_gray)
                flicker_index = np.mean(diff_img[mask]) / 255.0
                
                # Save flicker visualization
                self.save_step_image(diff_img, "20_flicker_diff", image_index, original_filename)
            else:
                flicker_index = 0.0
                diff_img = np.zeros_like(gray_img)
        
        # Update previous gray image for next iteration
        self.prev_gray = gray_img.copy()
        
        # Normalize values
        norm_brightness = brightness / self.reference_brightness if self.reference_brightness > 0 else 0
        norm_color_temp = 1.0 - abs(color_temp - self.reference_color_temp) / self.reference_color_temp if self.reference_color_temp > 0 else 0
        
        # Create visualization with measurements overlay
        result_img = img_masked.copy()
        
        # Add text overlay with measurements
        font = cv2.FONT_HERSHEY_SIMPLEX
        font_scale = 0.6
        color = (0, 255, 0)  # Green text
        thickness = 2
        
        text_lines = [
            f"Brightness: {brightness:.3f} ({norm_brightness:.3f})",
            f"Color Temp: {color_temp:.2f} ({norm_color_temp:.3f})",
            f"Flicker: {flicker_index:.5f}",
            f"Mask Area: {np.sum(mask)} pixels"
        ]
        
        y_offset = 30
        for i, line in enumerate(text_lines):
            cv2.putText(result_img, line, (10, y_offset + i * 25), 
                       font, font_scale, color, thickness)
        
        # Save final result image
        self.save_step_image(result_img, "21_final_result", image_index, original_filename)
        
        # Create and save composite visualization
        self.create_visualization_image(img, img_masked, hsv_img, lab_img, gray_img, 
                                      mask, image_index, original_filename)
        
        return {
            'brightness': brightness,
            'color_temp': color_temp,
            'flicker_index': flicker_index,
            'norm_brightness': norm_brightness,
            'norm_color_temp': norm_color_temp,
            'raw_img': img,
            'gray_img': gray_img,
            'diff_img': diff_img,
            'mask': mask
        }

    def estimate_lifespan(self, features):
        """
        Estimate remaining lifespan based on extracted features.
        
        Args:
            features (dict): Dictionary of extracted features
            
        Returns:
            float: Estimated remaining lifespan percentage (0-100%)
        """
        # Calculate individual health scores
        brightness_score = min(1.0, features['norm_brightness'] / self.brightness_threshold)
        
        # For color temperature, closer to reference is better (1.0 is perfect)
        color_temp_score = features['norm_color_temp']
        
        # For flicker, lower is better
        flicker_ratio = features['flicker_index'] / self.flicker_threshold
        flicker_score = max(0.0, 1.0 - flicker_ratio)
        
        # Calculate weighted health score
        health_score = (self.weights['brightness'] * brightness_score +
                        self.weights['color_temp'] * color_temp_score +
                        self.weights['flicker'] * flicker_score)
        
        # Clip to 0-100% range
        remaining_lifespan = min(max(health_score * 100, 0), 100)
        
        return remaining_lifespan

    def analyze_images(self, time_interval_days=1):
        """
        Analyze all images in the directory and track LED degradation over time.
        
        Args:
            time_interval_days (int): Simulated time interval between images in days
            
        Returns:
            list: Analysis history with timestamps and measurements
        """
        image_paths = self.load_images()
        if not image_paths:
            print("No images found in the directory.")
            return []
        
        
        for i, image_path in enumerate(image_paths):
            # Generate simulated timestamp (going backwards from current date)
            timestamp = datetime.fromtimestamp(os.path.getmtime(image_path))
            print(f"{image_path}: {timestamp}")
            
            print(f"\nAnalyzing image {i+1}/{len(image_paths)}: {os.path.basename(image_path)}")
            
            # Extract features from image
            features = self.extract_features(image_path, i)
            
            # Estimate remaining lifespan
            remaining_lifespan = self.estimate_lifespan(features)
            
            # Store data
            entry = {
                'timestamp': timestamp,
                'image_path': image_path,
                'brightness': features['brightness'],
                'norm_brightness': features['norm_brightness'],
                'color_temp': features['color_temp'],
                'norm_color_temp': features['norm_color_temp'],
                'flicker_index': features['flicker_index'],
                'remaining_lifespan': remaining_lifespan
            }
            
            self.history.append(entry)
            self.timestamps.append(timestamp)
            
            print(f"  Timestamp: {timestamp}")
            print(f"  Brightness: {features['brightness']:.3f} (Normalized: {features['norm_brightness']:.3f})")
            print(f"  Color Temperature: {features['color_temp']:.3f} (Normalized: {features['norm_color_temp']:.3f})")
            print(f"  Flicker Index: {features['flicker_index']:.5f}")
            print(f"  Estimated Remaining Lifespan: {remaining_lifespan:.1f}%")
            print("-" * 50)
            
        return self.history

    # Prepare x-axis as image filenames (without extension)

    def export_csv(self):
        """
        Export analysis history to a CSV file.
        
        Returns:
            str: Path to the exported CSV file
        """
        if not self.history:
            print("No data to export. Run analyze_images() first.")
            return None
        
        csv_path = os.path.join(self.output_dir, 'led_lifespan_data.csv')
        
        with open(csv_path, 'w', newline='') as csvfile:
            fieldnames = ['timestamp', 'image_path', 'brightness', 'norm_brightness', 
                          'color_temp', 'norm_color_temp', 'flicker_index', 'remaining_lifespan']
            writer = csv.DictWriter(csvfile, fieldnames=fieldnames)
            
            writer.writeheader()
            for entry in self.history:
                # Convert datetime to string
                entry_copy = entry.copy()
                entry_copy['timestamp'] = entry_copy['timestamp'].strftime('%Y-%m-%d %H:%M:%S')
                writer.writerow(entry_copy)
        
        print(f"Data exported to: {csv_path}")
        return csv_path

    def run_analysis(self, time_interval_days=1):
        """
        Run the complete analysis pipeline.
        
        Args:
            time_interval_days (int): Simulated time interval between images in days
            
        Returns:
            dict: Dictionary containing paths to output files
        """
        print("Starting LED bulb lifespan analysis...")
        print(f"Processing steps will be saved to: {self.steps_dir}")
        
        self.analyze_images(time_interval_days)
        
        print("\nGenerating visualization plots...")
        fig = self.plot_results()
        
        print("\nExporting data to CSV...")
        csv_path = self.export_csv()
        
        print("\nAnalysis complete!")
        
        if self.history:
            latest = self.history[-1]
            print(f"\nLatest Analysis Results ({latest['timestamp']}):")
            print(f"Brightness: {latest['norm_brightness']*100:.1f}%")
            print(f"Color Temperature Stability: {latest['norm_color_temp']*100:.1f}%")
            print(f"Flicker Index: {latest['flicker_index']*100:.3f}%")
            print(f"Estimated Remaining Lifespan: {latest['remaining_lifespan']:.1f}%")
        
        return {
            'csv': csv_path,
            'plot': os.path.join(self.output_dir, 'led_lifespan_trends.png'),
            'steps_dir': self.steps_dir
        }


def simulate_led_degradation(output_dir="sample_images", num_images=10, 
                             width=400, height=400, base_brightness=200,
                             brightness_decay_rate=0.05, color_temp_shift_rate=0.02,
                             flicker_increase_rate=0.01):
    """
    Generate simulated LED bulb images with progressive degradation.
    
    Args:
        output_dir (str): Directory to save generated images
        num_images (int): Number of images to generate
        width (int): Image width
        height (int): Image height
        base_brightness (int): Initial brightness value (0-255)
        brightness_decay_rate (float): Rate of brightness decay per image
        color_temp_shift_rate (float): Rate of color temperature shift per image
        flicker_increase_rate (float): Rate of flicker increase per image
        
    Returns:
        list: Paths to generated images
    """
    if not os.path.exists(output_dir):
        os.makedirs(output_dir)
    
    image_paths = []
    
    # Base color values (BGR)
    base_blue = 180
    base_green = 200
    base_red = 220  # Slightly warm white
    
    # Create circular mask for LED bulb shape
    Y, X = np.ogrid[:height, :width]
    center = (height // 2, width // 2)
    radius = min(width, height) // 3
    
    dist_from_center = ((Y - center[0]) ** 2 + (X - center[1]) ** 2)
    mask = dist_from_center <= radius ** 2
    
    # Add some noise to the initial state for realism
    np.random.seed(42)  # For reproducibility
    
    for i in range(num_images):
        # Create black background
        img = np.zeros((height, width, 3), dtype=np.uint8)
        
        # Apply degradation
        current_brightness = base_brightness * (1.0 - brightness_decay_rate * i)
        
        # Shift color temperature (get cooler over time)
        current_blue = min(255, base_blue * (1.0 + color_temp_shift_rate * i))
        current_green = base_green * (1.0 - color_temp_shift_rate * 0.5 * i)
        current_red = base_red * (1.0 - color_temp_shift_rate * i)
        
        # Create the LED bulb
        bulb_color = (
            int(current_blue), 
            int(current_green), 
            int(current_red)
        )
        
        # Apply the bulb color to the masked area
        img[mask] = bulb_color
        
        # Add random noise to simulate aging/flickering
        flicker_noise = np.random.normal(
            0, 
            flicker_increase_rate * (i + 1) * 15, 
            (height, width, 3)
        ).astype(np.int16)
        
        # Apply noise only to the bulb area
        temp_img = img.copy().astype(np.int16)
        temp_img[mask] += flicker_noise[mask]
        temp_img = np.clip(temp_img, 0, 255).astype(np.uint8)
        
        # Draw a faint bulb outline
        center_point = (width // 2, height // 2)
        cv2.circle(temp_img, center_point, radius, (100, 100, 100), 2)
        
        # Add some glow effect
        blurred = cv2.GaussianBlur(temp_img, (21, 21), 0)
        img = cv2.addWeighted(temp_img, 0.7, blurred, 0.3, 0)
        
        # Save the image
        image_path = os.path.join(output_dir, f"led_bulb_{i+1:02d}.png")
        cv2.imwrite(image_path, img)
        image_paths.append(image_path)
        
        print(f"Generated image {i+1}/{num_images}: {image_path}")
    
    return image_paths
